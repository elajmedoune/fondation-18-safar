=== SQL BONUS : INTERDIRE LES MEMBRES HORS CAMPAGNE (niveau base de donnees) ===
A executer dans Supabase Dashboard > SQL Editor > New query

create or replace function public.forbid_membre_hors_campagne()
returns trigger as $$
begin
  if not exists (
    select 1 from public.campagne_membres cm where cm.membre_id = new.id
  ) then
    raise exception 'Un membre doit etre rattache a une campagne.';
  end if;
  return new;
end;
$$ language plpgsql;

create trigger trg_membre_doit_avoir_campagne
after insert on public.membres
for each row execute function public.forbid_membre_hors_campagne();
